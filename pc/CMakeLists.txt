cmake_minimum_required(VERSION 3.16)
project(shuriken_controller LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(SHURIKEN_TIGARD "Build the Tigard controller and probe (requires libftdi1)" ON)
option(SHURIKEN_SCANNER_PROFILE "Enable per-sweep diagnostic timing" OFF)
option(SHURIKEN_SANITIZERS "Enable address/undefined sanitizers" OFF)
if(SHURIKEN_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_library(controller_hal controller_hal.cpp platform.cpp ../src/controller/time.cpp)
target_include_directories(controller_hal PUBLIC . ../include)
target_compile_options(controller_hal PRIVATE -Wall -Wextra -Werror)
file(GLOB XXHASH_HINTS "${CMAKE_CURRENT_SOURCE_DIR}/../.pio/libdeps/*/xxHash/include")
find_path(XXHASH_INCLUDE_DIR xxhash.h HINTS ${XXHASH_HINTS})
if(NOT XXHASH_INCLUDE_DIR)
  message(FATAL_ERROR "Install libxxhash-dev or build a PlatformIO controller to install xxHash")
endif()
add_library(scanner_runtime
  ../src/controller/scanner_runtime.cpp
  ../lib/wifi_common/wifi_dedupe.cpp
  ../lib/wifi_common/wifi_result_utils.cpp
  ../lib/log_format/log_format.cpp)
target_include_directories(scanner_runtime PUBLIC ../include ../lib/wifi_common
  ../lib/log_format PRIVATE ${XXHASH_INCLUDE_DIR})
target_compile_definitions(scanner_runtime PUBLIC SHURIKEN_PC_CONTROLLER
  SCANNER_SPI_CLOCK=10000000 PRIVATE XXH_INLINE_ALL)
target_compile_options(scanner_runtime PRIVATE -Wall -Wextra -Werror)
target_link_libraries(scanner_runtime PUBLIC controller_hal)
if(SHURIKEN_SCANNER_PROFILE)
  target_compile_definitions(scanner_runtime PUBLIC SCANNER_SWEEP_PROFILE=1)
endif()
file(GLOB GPS_HINTS "${CMAKE_CURRENT_SOURCE_DIR}/../.pio/libdeps/*/TinyGPSPlus/src")
find_path(TINYGPS_INCLUDE_DIR TinyGPS++.h HINTS ${GPS_HINTS})
if(NOT TINYGPS_INCLUDE_DIR)
  message(FATAL_ERROR "Build a PlatformIO controller to install TinyGPSPlus, or set TINYGPS_INCLUDE_DIR")
endif()
add_library(controller_application
  logging.cpp
  ../src/controller/application.cpp ../src/controller/runtime.cpp ../src/controller/scanner_update.cpp
  ../lib/controller_logging/controller_logging.cpp "${TINYGPS_INCLUDE_DIR}/TinyGPS++.cpp")
target_include_directories(controller_application PUBLIC compat ../include ../lib/controller_logging
  "${TINYGPS_INCLUDE_DIR}")
target_compile_definitions(controller_application PUBLIC ARDUINO=10805 SHURIKEN_PC_CONTROLLER)
target_compile_options(controller_application PRIVATE -Wall -Wextra)
target_link_libraries(controller_application PUBLIC scanner_runtime)
if(SHURIKEN_TIGARD)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(FTDI REQUIRED IMPORTED_TARGET libftdi1)
  add_executable(tigard-probe probe.cpp ftdi_link.cpp)
  target_link_libraries(tigard-probe PRIVATE controller_hal PkgConfig::FTDI)
  target_compile_options(tigard-probe PRIVATE -Wall -Wextra -Werror)
  add_executable(tigard-controller controller.cpp ftdi_link.cpp)
  target_link_libraries(tigard-controller PRIVATE controller_application PkgConfig::FTDI)
  target_compile_options(tigard-controller PRIVATE -Wall -Wextra -Werror)
endif()
